Practical 1: Quick Sort for Server Logs

Problem: Sort massive server logs chronologically using timestamps.

#include <stdio.h>

void swap(int* a, int* b) {
    int t = *a;
    *a = *b;
    *b = t;
}

int partition(int arr[], int low, int high) {
    int pivot = arr[high];
    int i = (low - 1);
    for (int j = low; j <= high - 1; j++) {
        if (arr[j] < pivot) {
            i++;
            swap(&arr[i], &arr[j]);
        }
    }
    swap(&arr[i + 1], &arr[high]);
    return (i + 1);
}

void quickSort(int arr[], int low, int high) {
    if (low < high) {
        int pi = partition(arr, low, high);
        quickSort(arr, low, pi - 1);
        quickSort(arr, pi + 1, high);
    }
}

int main() {
    int timestamps[] = {1672531199, 1672444799, 1672617599};
    int n = sizeof(timestamps) / sizeof(timestamps[0]);
    quickSort(timestamps, 0, n - 1);
    printf("Sorted Timestamps:\n");
    for (int i = 0; i < n; i++) printf("%d\n", timestamps[i]);
    return 0;
}
